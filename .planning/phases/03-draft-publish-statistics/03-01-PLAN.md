```typescript
export async function publishDraft(): Promise<{ success: true; count: number } | { success: false; error: string }> {
  const session = await auth()
  if (!session?.user) throw new Error("Not authenticated")

  const { start, end } = getWindowBounds()
  const startStr = format(start, "yyyy-MM-dd")
  const endStr = format(end, "yyyy-MM-dd")

  const result = await db.update(scheduleEntries)
    .set({ status: "published" })
    .where(
      and(
        eq(scheduleEntries.status, "draft"),
        gte(scheduleEntries.day, startStr),
        lte(scheduleEntries.day, endStr),
      )
    )
    .returning({ id: scheduleEntries.id })

  return { success: true, count: result.length }
}
```

Key details:
- Uses `getWindowBounds()` (already exists in `src/lib/schedule/generate-default.ts`) to scope the update to the current 84-day window (per D-01: all draft entries in window)
- Returns the count of updated rows for the confirmation toast
- Auth guard follows existing pattern from `toggleCell`
- The bulk UPDATE triggers Supabase Realtime CDC events for each changed row — existing realtime handler in schedule-table.tsx already handles per-entry status updates

1. Create `src/components/schedule/publish-button.tsx` — a "use client" component:

```typescript
"use client"

import { useState } from "react"
import { toast } from "sonner"
import { Button } from "@/components/ui/button"
import {
  Dialog,
  DialogContent,
  DialogDescription,
  DialogFooter,
  DialogHeader,
  DialogTitle,
  DialogTrigger,
} from "@/components/ui/dialog"
import { publishDraft } from "@/actions/schedule"
import type { DateWindow } from "@/lib/schedule/types"
import { format, parseISO } from "date-fns"

interface PublishButtonProps {
  initialData: DateWindow
}

export function PublishButton({ initialData }: PublishButtonProps) {
  const [open, setOpen] = useState(false)
  const [publishing, setPublishing] = useState(false)

  // Derive draft count and date range from the schedule data (per Claude's discretion — client-side from already-loaded data)
  const draftCells = initialData.days.flatMap(day =>
    day.cells.filter(cell => cell.status === "draft")
  )
  const draftCount = draftCells.length

  // Find date range of draft entries
  const draftDays = initialData.days.filter(day =>
    day.cells.some(cell => cell.status === "draft")
  )
  const firstDraftDate = draftDays[0]?.date
  const lastDraftDate = draftDays[draftDays.length - 1]?.date

  const dateRangeLabel = firstDraftDate && lastDraftDate
    ? `${format(parseISO(firstDraftDate), "d MMM")} \u2013 ${format(parseISO(lastDraftDate), "d MMM yyyy")}`
    : ""

  async function handlePublish() {
    setPublishing(true)
    try {
      const result = await publishDraft()
      if (result.success) {
        toast.success(`Published ${result.count} entries`)
        setOpen(false)
      } else {
        toast.error(result.error)
      }
    } catch {
      toast.error("Failed to publish. Please try again.")
    } finally {
      setPublishing(false)
    }
  }

  // Per D-08: disabled when no draft entries exist
  if (draftCount === 0) {
    return (
      <Button variant="outline" size="sm" disabled>
        Publish
      </Button>
    )
  }

  return (
    <Dialog open={open} onOpenChange={setOpen}>
      <DialogTrigger asChild>
        <Button variant="default" size="sm">
          Publish
        </Button>
      </DialogTrigger>
      <DialogContent>
        <DialogHeader>
          <DialogTitle>Publish Schedule</DialogTitle>
          <DialogDescription>
            Publish {draftCount} draft entries ({dateRangeLabel})?
            This will lock the schedule and sync to Google Calendar.
          </DialogDescription>
        </DialogHeader>
        <DialogFooter>
          <Button variant="outline" onClick={() => setOpen(false)} disabled={publishing}>
            Cancel
          </Button>
          <Button onClick={handlePublish} disabled={publishing}>
            {publishing ? "Publishing..." : "Confirm"}
          </Button>
        </DialogFooter>
      </DialogContent>
    </Dialog>
  )
}
```

Key decisions per CONTEXT.md:
- Per D-02: Dialog shows count + date range (e.g. "Publish 48 draft entries (6 Jan - 29 Mar)?")
- Per D-03: Uses shadcn/ui Dialog component
- Per D-07: Button placed in Header, right side
- Per D-08: Button disabled when draftCount === 0
- Draft count derived client-side from already-loaded schedule data (Claude's discretion)
