1. Create `src/lib/schedule/stats.ts`:

```typescript
import type { ScheduleDay, ParentId } from "./types"
import { isWeekend, parseISO, getDay } from "date-fns"

export interface ChildStats {
  childName: string
  father: number   // total days with father
  mother: number   // total days with mother
  soloFather: number  // days child is alone with father (no other children present with father)
  soloMother: number  // days child is alone with mother
}

export interface ParentFreeStats {
  parentId: ParentId
  parentName: string
  childFreeDays: number      // days with zero children
  childFreeWeekends: number  // full Sat+Sun weekends with zero children
}

export interface ScheduleStats {
  childStats: ChildStats[]
  parentFreeStats: ParentFreeStats[]
}

export function computeStats(
  days: ScheduleDay[],
  parents: Array<{ id: ParentId; name: string }>,
): ScheduleStats {
  const childNames = days[0]?.cells.map(c => c.childName) ?? []

  // Per-child: count days with each parent
  const childStatsMap = new Map<string, { father: number; mother: number; soloFather: number; soloMother: number }>()
  for (const name of childNames) {
    childStatsMap.set(name, { father: 0, mother: 0, soloFather: 0, soloMother: 0 })
  }

  // Per-parent: track child-free days
  const parentFreeDaysMap = new Map<ParentId, Set<string>>()
  for (const p of parents) {
    parentFreeDaysMap.set(p.id, new Set())
  }

  for (const day of days) {
    // Which children are with which parent on this day
    const withFather = day.cells.filter(c => c.parentId === "father").map(c => c.childName)
    const withMother = day.cells.filter(c => c.parentId === "mother").map(c => c.childName)

    // STAT-01: days per child per parent
    for (const cell of day.cells) {
      const stats = childStatsMap.get(cell.childName)!
      stats[cell.parentId]++
    }

    // STAT-02: solo days — child is alone with parent (only child present with that parent)
    if (withFather.length === 1) {
      childStatsMap.get(withFather[0])!.soloFather++
    }
    if (withMother.length === 1) {
      childStatsMap.get(withMother[0])!.soloMother++
    }

    // STAT-03: child-free days
    if (withFather.length === 0) {
      parentFreeDaysMap.get("father")!.add(day.date)
    }
    if (withMother.length === 0) {
      parentFreeDaysMap.get("mother")!.add(day.date)
    }
  }

  // STAT-04: child-free weekends (full Sat+Sun)
  // Group Saturdays and check if the following Sunday is also child-free
  function countChildFreeWeekends(freeDays: Set<string>): number {
    let count = 0
    for (const dateStr of freeDays) {
      const date = parseISO(dateStr)
      if (getDay(date) === 6) { // Saturday
        // Check if the next day (Sunday) is also free
        const sundayStr = days.find(d => {
          const dDate = parseISO(d.date)
          return getDay(dDate) === 0 && dDate.getTime() - date.getTime() === 86400000
        })?.date
        if (sundayStr && freeDays.has(sundayStr)) {
          count++
        }
      }
    }
    return count
  }

  const childStats: ChildStats[] = childNames.map(name => ({
    childName: name,
    ...childStatsMap.get(name)!,
  }))

  const parentFreeStats: ParentFreeStats[] = parents.map(p => ({
    parentId: p.id,
    parentName: p.name,
    childFreeDays: parentFreeDaysMap.get(p.id)!.size,
    childFreeWeekends: countChildFreeWeekends(parentFreeDaysMap.get(p.id)!),
  }))

  return { childStats, parentFreeStats }
}
```

Key details per STAT requirements:
- STAT-01: `childStats[].father` and `.mother` — total days per child per parent
- STAT-02: `childStats[].soloFather` and `.soloMother` — solo means that child is the ONLY child with that parent on that day
- STAT-03: `parentFreeStats[].childFreeDays` — days where zero children are with that parent
- STAT-04: `parentFreeStats[].childFreeWeekends` — full Sat+Sun where parent has zero children
- STAT-05: No status filter — function operates on all cells regardless of draft/published (per D-06)

1. Create `src/components/schedule/stats-panel.tsx`:

```typescript
"use client"

import { useMemo } from "react"
import { computeStats } from "@/lib/schedule/stats"
import config from "@/config/app"
import type { ScheduleDay } from "@/lib/schedule/types"

interface StatsPanelProps {
  days: ScheduleDay[]
}

export function StatsPanel({ days }: StatsPanelProps) {
  const stats = useMemo(() => computeStats(days, config.parents), [days])

  // Parent name lookup
  const parentName = (id: string) =>
    config.parents.find(p => p.id === id)?.name ?? id

  return (
    <div className="border rounded-lg p-3 mb-4 bg-muted/30 text-sm space-y-1">
      {stats.childStats.map(child => (
        <div key={child.childName} className="flex items-center gap-4">
          <span className="font-medium w-16">{child.childName}:</span>
          <span className="text-blue-700">{parentName("father")} {child.father}d</span>
          <span className="text-muted-foreground">/</span>
          <span className="text-rose-700">{parentName("mother")} {child.mother}d</span>
          <span className="text-muted-foreground ml-2">
            (solo: {parentName("father")} {child.soloFather}d / {parentName("mother")} {child.soloMother}d)
          </span>
        </div>
      ))}
      <div className="border-t pt-1 mt-1 flex items-center gap-4">
        <span className="font-medium w-16">Free:</span>
        {stats.parentFreeStats.map(ps => (
          <span key={ps.parentId} className={ps.parentId === "father" ? "text-blue-700" : "text-rose-700"}>
            {ps.parentName} {ps.childFreeDays}d ({ps.childFreeWeekends} wknd{ps.childFreeWeekends !== 1 ? "s" : ""})
          </span>
        ))}
      </div>
    </div>
  )
}
```

```typescript
"use client"

import { useRef, useCallback } from "react"
import { ScheduleTable } from "./schedule-table"
import { StatsPanel } from "./stats-panel"
import { RealtimeProvider } from "./realtime-provider"
import type { DateWindow, ParentId } from "@/lib/schedule/types"

type EntryUpdate = {
  id: string
  childId: string
  day: string
  parentId: ParentId
  status: "draft" | "published"
  notes: string | null
}

export function ScheduleWithRealtime({ initialData }: { initialData: DateWindow }) {
  const realtimeRef = useRef<((entry: EntryUpdate) => void) | null>(null)

  const handleEntryChange = useCallback((entry: EntryUpdate) => {
    realtimeRef.current?.(entry)
  }, [])

  return (
    <RealtimeProvider onEntryChange={handleEntryChange}>
      <ScheduleTable
        initialData={initialData}
        realtimeRef={realtimeRef}
        renderAbove={(days) => <StatsPanel days={days} />}
      />
    </RealtimeProvider>
  )
}
```
