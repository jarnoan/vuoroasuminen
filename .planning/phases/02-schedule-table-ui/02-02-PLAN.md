   - Export a `setDays` ref or callback so Plan 03 can inject realtime updates (use a ref: `const daysRef = useRef(days)` and expose a `updateFromRealtime` function via a React context or direct prop -- for now, expose via a module-level setter that Plan 03 will replace with Supabase subscription)

2. **Optimistic cell toggle (D-05):**
   ```typescript
   async function handleToggle(entryId: string, newParentId: ParentId) {
     // Optimistic: update local state immediately
     setDays(prev => prev.map(day => ({
       ...day,
       cells: day.cells.map(cell =>
         cell.entryId === entryId ? { ...cell, parentId: newParentId } : cell
       )
     })))

     try {
       await toggleCell(entryId, newParentId)
     } catch {
       // Revert on failure
       setDays(prev => prev.map(day => ({
         ...day,
         cells: day.cells.map(cell =>
           cell.entryId === entryId
             ? { ...cell, parentId: newParentId === "father" ? "mother" : "father" }
             : cell
         )
       })))
       toast.error("Failed to save change. Please try again.")
     }
   }
   ```

3. **Notes save handler (D-08):**
   ```typescript
   async function handleNoteSave(entryId: string, notes: string) {
     // Update local state
     setDays(prev => prev.map(day =>
       day.notesEntryId === entryId ? { ...day, notes } : day
     ))
     try {
       await saveNotes(entryId, notes)
     } catch {
       toast.error("Failed to save note.")
     }
   }
   ```

4. **Auto-scroll to today on mount (D-03):**

   useEffect(() => {
