**Create `src/lib/supabase/client.ts`:**

Browser-side Supabase client for Realtime subscriptions. Uses `@supabase/supabase-js` (already installed).

```typescript
import { createClient } from "@supabase/supabase-js"

let client: ReturnType<typeof createClient> | null = null

export function createBrowserClient() {
  if (client) return client

  const url = process.env.NEXT_PUBLIC_SUPABASE_URL!
  const anonKey = process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!

  client = createClient(url, anonKey, {
    realtime: {
      params: { eventsPerSecond: 10 },
    },
  })

  return client
}
```

Note: This does NOT use `@supabase/ssr` because we only need the Realtime subscription (no Supabase Auth -- we use Auth.js). A plain `createClient` from `@supabase/supabase-js` is correct for browser-side Realtime channels.

**Create `src/components/schedule/realtime-provider.tsx`:**

A `"use client"` wrapper component that subscribes to Supabase Postgres Changes on `schedule_entries` and calls a callback to update the schedule table state.

```typescript
"use client"

import { useEffect, useRef } from "react"
import { createBrowserClient } from "@/lib/supabase/client"
import type { ParentId } from "@/lib/schedule/types"

interface RealtimePayload {
  new: {
    id: string
    child_id: string
    day: string
    parent_id: string
    status: string
    notes: string | null
  }
}

interface RealtimeProviderProps {
  children: React.ReactNode
  onEntryChange: (entry: {
    id: string
    childId: string
    day: string
    parentId: ParentId
    status: "draft" | "published"
    notes: string | null
  }) => void
}

export function RealtimeProvider({ children, onEntryChange }: RealtimeProviderProps) {
  const onEntryChangeRef = useRef(onEntryChange)
  onEntryChangeRef.current = onEntryChange

  useEffect(() => {
    const supabase = createBrowserClient()

    const channel = supabase
      .channel("schedule-changes")
      .on(
        "postgres_changes",
        {
          event: "*",        // INSERT and UPDATE
          schema: "public",
          table: "schedule_entries",
        },
        (payload: { new: Record<string, unknown> }) => {
          const row = payload.new as RealtimePayload["new"]
          onEntryChangeRef.current({
            id: row.id,
            childId: row.child_id,
            day: row.day,
            parentId: row.parent_id as ParentId,
            status: row.status as "draft" | "published",
            notes: row.notes,
          })
        }
      )
      .subscribe()

    return () => {
      supabase.removeChannel(channel)
    }
  }, [])

  return <>{children}</>
}
```

Key design: The payload column names are snake_case (Postgres convention) while our TypeScript types use camelCase. The provider maps between them.

const handleRealtimeEntry = useCallback((entry: {
  id: string; childId: string; day: string; parentId: ParentId; status: "draft" | "published"; notes: string | null
}) => {
  setDays(prev => prev.map(day => {
    if (day.date !== entry.day) return day
    return {
      ...day,
      cells: day.cells.map(cell =>
        cell.entryId === entry.id || (cell.childId === entry.childId && cell.entryId === null)
          ? { ...cell, entryId: entry.id, parentId: entry.parentId, status: entry.status }
          : cell
      ),
      notes: entry.notes ?? day.notes,
      notesEntryId: day.notesEntryId === entry.id ? entry.id : day.notesEntryId,
    }
  }))
}, [])

**Create `src/components/schedule/schedule-with-realtime.tsx`:**

A thin client wrapper connecting RealtimeProvider to ScheduleTable via a ref:

```tsx
"use client"

import { useRef, useCallback } from "react"
import { ScheduleTable } from "./schedule-table"
import { RealtimeProvider } from "./realtime-provider"
import type { DateWindow, ParentId } from "@/lib/schedule/types"

type EntryUpdate = {
  id: string; childId: string; day: string; parentId: ParentId;
  status: "draft" | "published"; notes: string | null
}

export function ScheduleWithRealtime({ initialData }: { initialData: DateWindow }) {
  const realtimeRef = useRef<((entry: EntryUpdate) => void) | null>(null)

  const handleEntryChange = useCallback((entry: EntryUpdate) => {
    realtimeRef.current?.(entry)
  }, [])

  return (
    <RealtimeProvider onEntryChange={handleEntryChange}>
      <ScheduleTable initialData={initialData} realtimeRef={realtimeRef} />
    </RealtimeProvider>
  )
}
```
